  function beep(freq, duration, delay, type) {

    if (!soundOn) return;

    var ctx = ensureAudio();

    if (!ctx) return;

    var t0 = ctx.currentTime + (delay || 0);

    var osc = ctx.createOscillator();

    var gain = ctx.createGain();

    osc.type = type || 'sine';

    osc.frequency.setValueAtTime(freq, t0);

    gain.gain.setValueAtTime(0.0001, t0);

    gain.gain.exponentialRampToValueAtTime(0.22, t0 + 0.02);

    gain.gain.exponentialRampToValueAtTime(0.0001, t0 + duration);

    osc.connect(gain).connect(ctx.destination);

    osc.start(t0);

    osc.stop(t0 + duration + 0.03);

  }

  function redrawIllustration(canvas, strokes, opacity) {

    ctx.lineJoin = 'round';

      if (!s.points || s.points.length === 0) return;

      ctx.strokeStyle = s.color;

      ctx.lineWidth = s.size;

      ctx.beginPath();

      ctx.moveTo(s.points[0].x, s.points[0].y);

      for (var i = 1; i < s.points.length; i++) ctx.lineTo(s.points[i].x, s.points[i].y);

      ctx.stroke();

    });

    ctx.globalAlpha = 1;

  }


  // 点を間引いてデータ量を減らす(見た目はほぼそのまま)

  function simplifyPoints(points, minDist) {

    if (points.length <= 2) return points;

    var out = [points[0]];

    for (var i = 1; i < points.length; i++) {

      if (dist(out[out.length - 1], points[i]) >= minDist) out.push(points[i]);

    }

    if (out[out.length - 1] !== points[points.length - 1]) out.push(points[points.length - 1]);

    return out;

  }

  function bytesToBase64Url(bytes) {

    var binary = '';

    var chunk = 0x8000;

    for (var i = 0; i < bytes.length; i += chunk) binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunk));

    var b64 = btoa(binary);

    return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');

  }

  function base64UrlToBytes(str) {

    str = str.replace(/-/g, '+').replace(/_/g, '/');

    while (str.length % 4) str += '=';

    var binary = atob(str);

    var bytes = new Uint8Array(binary.length);

    for (var i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);

    return bytes;

  }

  function fromCompact(compact) {

    var dots = [];

    var d = compact.d || [];

    for (var i = 0; i < d.length; i += 2) dots.push({ x: d[i], y: d[i + 1] });

    var strokes = (compact.s || []).map(function (arr) {

      var colorIdx = arr[0], sizeIdx = arr[1];

      var points = [];

      for (var i = 2; i < arr.length; i += 2) points.push({ x: arr[i], y: arr[i + 1] });

      return { color: DRAW_COLORS[colorIdx] || DRAW_COLORS[0], size: DRAW_SIZES[sizeIdx] || 8, points: points };

    });

    return { title: compact.t || '', charSet: CHARSET_CODE_REV[compact.c] || 'hiragana', showGuideLine: !!compact.g, dots: dots, illustration: strokes };

  }

  function encodePuzzleToPayload(puzzle) {

    var json = JSON.stringify(toCompact(puzzle));

    var flag, bytes;

    if (window.pako) {

      try { bytes = pako.deflate(json); flag = '1'; }

      catch (e) { bytes = new TextEncoder().encode(json); flag = '0'; }

    } else {

      bytes = new TextEncoder().encode(json); flag = '0';

    }

    return flag + bytesToBase64Url(bytes);

  }


  function decodePayloadToPuzzle(payload) {

    var flag = payload.charAt(0);

    var b64 = payload.slice(1);

    var bytes = base64UrlToBytes(b64);

    var json;

    if (flag === '1' && window.pako) json = pako.inflate(bytes, { to: 'string' });

    else json = new TextDecoder().decode(bytes);

    var compact = JSON.parse(json);

    return fromCompact(compact);

  }

  }


  function extractPayloadFromText(text) {

    if (!text) return null;

    var idx = text.indexOf('#p=');

    if (idx >= 0) return text.slice(idx + 3);

    var qidx = text.indexOf('?p=');

    if (qidx >= 0) return text.slice(qidx + 3);

    // コードの文字列そのものが渡された場合

    if (/^[01][A-Za-z0-9_-]{10,}$/.test(text.trim())) return text.trim();

    return null;

  }

  function startCamera() {

    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {

      $('#scanStatus').textContent = 'このブラウザは カメラに たいおうしていないよ。したの らんに リンクを はりつけてね';

      return;

    }

    navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: 'environment' } } })

      .then(function (stream) {

        scanStream = stream;

        scanVideo.srcObject = stream;

        scanVideo.setAttribute('playsinline', true);

        scanVideo.play();

        $('#scanStatus').textContent = 'QRコードを カメラに みせてね';

        scanRafId = requestAnimationFrame(scanLoop);

      })

      .catch(function () {

        $('#scanStatus').textContent = 'カメラが つかえなかったよ。したの らんに リンクを はりつけてね';

      });

  }

  function stopCamera() {

    if (scanRafId) { cancelAnimationFrame(scanRafId); scanRafId = null; }

    if (scanStream) { scanStream.getTracks().forEach(function (t) { t.stop(); }); scanStream = null; }

    scanVideo.srcObject = null;

  }


  function scanLoop() {

    if (!scanStream) return;

    if (scanVideo.readyState === scanVideo.HAVE_ENOUGH_DATA && window.jsQR) {

      scanCanvas.width = scanVideo.videoWidth;

      scanCanvas.height = scanVideo.videoHeight;

      var ctx = scanCanvas.getContext('2d');

      ctx.drawImage(scanVideo, 0, 0, scanCanvas.width, scanCanvas.height);

      try {

        var imageData = ctx.getImageData(0, 0, scanCanvas.width, scanCanvas.height);

        var code = jsQR(imageData.data, imageData.width, imageData.height);

        if (code && code.data) { handleScannedText(code.data); return; }

      } catch (e) { /* ignore frame errors */ }

    }

    scanRafId = requestAnimationFrame(scanLoop);

  }


  function handleScannedText(text) {

    var payload = extractPayloadFromText(text);

    if (!payload) { $('#scanStatus').textContent = 'それは てんつなぎの QRじゃ ないみたい...もういちど!'; scanRafId = requestAnimationFrame(scanLoop); return; }

    try {

      var puzzle = decodePayloadToPuzzle(payload);

      if (!puzzle.dots || puzzle.dots.length < 2) throw new Error('empty');

      playScanSound();

      closeScanModal();

      loadPuzzleIntoPlay(puzzle, true);

      switchTab('play');

      showToast('よみとれたよ！あそぼう🎉');

    } catch (e) {

      $('#scanStatus').textContent = 'よみとれなかったよ...もういちど ためしてね';

      scanRafId = requestAnimationFrame(scanLoop);

    }

  }

    if (!puzzle) return;

  function renderSequenceTrack() {

    var track = $('#sequenceTrack');

    track.innerHTML = '';

    if (!playState.puzzle) return;

    var chars = playCharList();

    playState.puzzle.dots.forEach(function (dot, i) {

      var item = document.createElement('div');

      var cls = 'seq-item';

      if (i < playState.currentIndex + 1) cls += ' done';

      else if (i === playState.currentIndex + 1) cls += ' current';

      item.className = cls;

      item.textContent = chars[i] || '?';

      track.appendChild(item);

    });

  }


  function renderPlaySvg() {

    playSvg.innerHTML = '';

    if (!playState.puzzle) return;

    var dots = playState.puzzle.dots;

    var chars = playCharList();


    if (playState.hintOn && dots.length > 1) {

      var d = 'M ' + dots.map(function (p) { return p.x + ' ' + p.y; }).join(' L ');

      playSvg.appendChild(svgEl('path', { d: d, class: 'guide-line' }));

    }


    for (var i = 0; i < playState.currentIndex; i++) {

      var a = dots[i], b = dots[i + 1];

      playSvg.appendChild(svgEl('line', { x1: a.x, y1: a.y, x2: b.x, y2: b.y, class: 'segment-line', stroke: segColor(i) }));

    }


    dots.forEach(function (dot, i) {

      var g = svgEl('g', { class: 'dot-group', 'data-index': i });

      var isDone = i <= playState.currentIndex;

      var isStart = i === playState.currentIndex;

      var isNextTarget = i === playState.currentIndex + 1;


      var fill = isDone ? '#51CF66' : '#fff';

      var stroke = isDone ? '#2FBF71' : '#FF8FAB';


      if (isStart && i < dots.length - 1) playSvg.appendChild(svgEl('circle', { cx: dot.x, cy: dot.y, r: 30, fill: '#FF8FAB', class: 'dot-pulse' }));


      var c = svgEl('circle', { cx: dot.x, cy: dot.y, r: 22, fill: fill, stroke: stroke, 'stroke-width': 4, class: 'dot-circle' });

      var t = svgEl('text', { x: dot.x, y: dot.y + 1, class: 'dot-label', 'font-size': 18, fill: isDone ? '#fff' : '#4A3F35' });

      t.textContent = chars[i] || '?';

      g.appendChild(c); g.appendChild(t);


      if (isNextTarget) {

        var arrow = svgEl('text', { x: dot.x, y: dot.y - 34, 'font-size': 26, 'text-anchor': 'middle', class: 'arrow-bounce' });

        arrow.textContent = '⬇️';

        playSvg.appendChild(arrow);

      }

      playSvg.appendChild(g);

    });

  }


  function updateEncouragement() {

    if (!playState.puzzle) { $('#encouragementText').textContent = ''; return; }

    var total = playState.puzzle.dots.length - 1;

    if (total <= 0) return;

    var pct = playState.currentIndex / total;

    var msg = 'がんばって！';

    if (pct >= 1) msg = 'かんせい！🎉';

    else if (pct >= 0.75) msg = 'あと すこし！💪';

    else if (pct >= 0.4) msg = 'いいかんじ！😊';

    else if (pct > 0) msg = 'その ちょうし！✨';

    $('#encouragementText').textContent = msg;

  }

  var tempLine = null;


  playWrap.addEventListener('pointerdown', function (evt) {

    if (!playState.puzzle) return;

    var dots = playState.puzzle.dots;

    if (playState.currentIndex >= dots.length - 1) return;


    var pt = pointerToLogical(evt, playWrap);

    var idx = findNearestDotIndex(dots, pt, 30);


    if (idx === playState.currentIndex) {

      playState.dragging = true;

      playState.dragFromIndex = idx;

      playState.dragStartPt = { x: dots[idx].x, y: dots[idx].y };

      try { evt.target.setPointerCapture(evt.pointerId); } catch (e) {}

      tempLine = svgEl('line', { x1: dots[idx].x, y1: dots[idx].y, x2: dots[idx].x, y2: dots[idx].y, class: 'temp-drag-line' });

      playSvg.appendChild(tempLine);

    } else if (idx >= 0) {

      wiggleDot(idx); wiggleDot(playState.currentIndex);

    }

  });


  playWrap.addEventListener('pointermove', function (evt) {

    if (!playState.dragging || !tempLine) return;

    var pt = pointerToLogical(evt, playWrap);

    tempLine.setAttribute('x2', pt.x); tempLine.setAttribute('y2', pt.y);

  });


  function endPlayDrag(evt) {

    if (!playState.dragging) return;

    var dots = playState.puzzle.dots;

    var pt = pointerToLogical(evt, playWrap);

    if (tempLine) { tempLine.remove(); tempLine = null; }


    var movedDist = dist(pt, playState.dragStartPt);

    var fromIdx = playState.dragFromIndex;

    var targetIdx = fromIdx + 1;

    playState.dragging = false;

    playState.dragFromIndex = -1;

    if (movedDist < MIN_DRAG_DIST) return;


    var targetDot = dots[targetIdx];

    if (targetDot && dist(pt, targetDot) <= HIT_RADIUS) {

      playState.currentIndex = targetIdx;

      playSuccessSound();

      renderSequenceTrack(); renderPlaySvg(); updateEncouragement();

      if (playState.currentIndex >= dots.length - 1) setTimeout(triggerClear, 260);

    } else {

      playFailSound(); shakeDot(fromIdx);

    }

  }

  playWrap.addEventListener('pointerup', endPlayDrag);

  playWrap.addEventListener('pointercancel', function () {

    if (tempLine) { tempLine.remove(); tempLine = null; }

    playState.dragging = false; playState.dragFromIndex = -1;

  });

  function tryLoadFromHash() {

    var hash = location.hash || '';

    var idx = hash.indexOf('#p=');

    if (idx !== 0) return false;

    var payload = hash.slice(3);

    try {

      var puzzle = decodePayloadToPuzzle(payload);

      if (!puzzle.dots || puzzle.dots.length < 2) return false;

      loadPuzzleIntoPlay(puzzle, true);

      switchTab('play');

      return true;

    } catch (e) { return false; }

  }
